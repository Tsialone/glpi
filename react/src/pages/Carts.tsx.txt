import { useEffect, useRef, useState } from "react";
import { useCart } from "../hooks/carts.hook";
import type { ICart } from "../types/carts/Cart";

export default function Carts() {
    const { getCarts } = useCart();
    const [carts, setCarts] = useState<ICart[]>([]);
    const columns = useRef(['Client', 'Date', 'Total']);
    async function updateCarts() {
        const carts = await getCarts();
        setCarts(carts);
    }
    useEffect(() => {
        updateCarts()
    }, []);
    return (
        <div>
            <h2>Carts</h2>
            {carts.length > 0 &&
                <table border={1}>
                    <thead>
                        <tr>
                            {
                                columns.current.map((col, index) =>
                                (
                                    <td key={index}> {col} </td>
                                )
                                )
                            }
                        </tr>
                    </thead>
                    <tbody>
                        {
                            carts.map(c => (
                                <tr key={c.id} >
                                    <td>
                                        {c.customer?.lastname}
                                    </td>
                                    <td>
                                        {c.date_upd}
                                    </td>
                                    <td>
                                        {c.total}
                                    </td>
                                </tr>
                            ))
                        }
                    </tbody>
                </table>

            }

        </div>
    )
}
